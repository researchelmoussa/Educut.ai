\subsection*{Exercise 1 --- Multiple Choice \hfill {\normalsize\color{fnavy}30 points}}
\meta{(Level 3 --- Hard)\quad 5 questions $\times$ 6 points --- exactly one correct answer each}
\begin{enumerate}[label=\textbf{Q\arabic*.},leftmargin=*,itemsep=10pt]
\qcm{For $f(x)=\dfrac{1}{x+1}$, the simplified difference quotient $\dfrac{f(x+h)-f(x)}{h}$ is:}
  {$\dfrac{-1}{(x+1)(x+h+1)}$}{$\dfrac{1}{(x+1)(x+h+1)}$}{$\dfrac{-h}{(x+1)(x+h+1)}$}{$\dfrac{-1}{(x+1)^2}$}
\qcm{If $y=\dfrac{x^2-4\sqrt{x}}{x}$, then $y'(4)$ equals:}
  {$\tfrac54$}{$\tfrac34$}{$0$}{$2$}
\qcm{Which function is an antiderivative of $f(x)=\dfrac{e^{2x}-3}{e^{x}}$?}
  {$e^{x}-3e^{-x}$}{$e^{x}+3e^{-x}+7$}{$e^{x}-3e^{x}$}{$\dfrac{\frac12 e^{2x}-3x}{e^{x}}$}
\qcm{$\displaystyle\int_{4}^{1}\left(3\sqrt{x}-2\right)dx=$}
  {$8$}{$-8$}{$14$}{$-14$}
\qcm{Let $F'(x)=\dfrac{2}{x}-6x^2$ for $x>0$, with $F(1)=0$. Then $F(e)=$}
  {$4-2e^3$}{$2-2e^3$}{$4+2e^3$}{$-2e^3$}
\end{enumerate}

\subsection*{Exercise 2 --- Differentiation \hfill {\normalsize\color{fnavy}40 points}}
\meta{(Level 2--3 --- Medium to Hard)\quad 5 questions $\times$ 8 points --- rewrite before you differentiate}
\begin{enumerate}[label=\textbf{(\alph*)},leftmargin=*,itemsep=4pt]
\item Find $f'(x)$ for $f(x)=3x^4-5\sqrt[3]{x^2}+\dfrac{7}{x^3}-\pi^2$. \pts{8}\answerspace{3.2cm}
\item Find $g'(x)$ for $g(x)=\left(\sqrt{x}+2\right)\left(x-\dfrac{1}{\sqrt{x}}\right)$. \pts{8}\answerspace{3.2cm}
\item Find $h'(x)$ for $h(x)=\dfrac{x^3-4x+6}{x^2}$. \pts{8}\answerspace{3.2cm}
\item Find $k'(t)$ for $k(t)=(2t-1)^3$. \emph{(Expand first.)} \pts{8}\answerspace{3.2cm}
\item Let $f(x)=\dfrac{(x-2)^2}{\sqrt{x}}$. Find $f'(x)$, then the equation of the tangent line to the graph of $f$ at $x=4$. \pts{8}\answerspace{4cm}
\end{enumerate}
